Retrieve the billing overview for your organization, including the payment method, upcoming charge, and subscriptions.

<RequestExample>
```bash cURL
curl -X GET 'https://api.cloud.blnkfinance.com/billing/overview' \
  -H 'Authorization: Bearer YOUR_API_KEY'
```
</RequestExample>

<ResponseField name="email" type="string" required>
Billing email.
</ResponseField>

<ResponseField name="name" type="string" required>
Organization or account name.
</ResponseField>

<ResponseField name="address" type="object">
Billing address.
<Expandable>
  <ResponseField name="line1" type="string">
  Street address.
  </ResponseField>
  <ResponseField name="city" type="string">
  City.
  </ResponseField>
  <ResponseField name="state" type="string">
  State or region.
  </ResponseField>
  <ResponseField name="postal_code" type="string">
  Postal code.
  </ResponseField>
  <ResponseField name="country" type="string">
  Country code, such as `US`.
  </ResponseField>
</Expandable>
</ResponseField>

<ResponseField name="card" type="object">
Payment method summary. `brand` can be a card network or a wallet such as `Link`. Some methods return empty `last4` and `0` expiry.
<Expandable>
  <ResponseField name="brand" type="string">
  Payment method brand, such as `visa` or `Link`.
  </ResponseField>
  <ResponseField name="last4" type="string">
  Last four digits of the card. Can be empty.
  </ResponseField>
  <ResponseField name="exp_month" type="integer">
  Expiration month. Can be `0`.
  </ResponseField>
  <ResponseField name="exp_year" type="integer">
  Expiration year. Can be `0`.
  </ResponseField>
</Expandable>
</ResponseField>

<ResponseField name="next_billing_date" type="integer">
Unix timestamp for the next charge.
</ResponseField>

<ResponseField name="next_billing_amount" type="integer">
Upcoming charge in the smallest currency unit (for example, cents).
</ResponseField>

<ResponseField name="currency" type="string">
ISO currency code, such as `usd`.
</ResponseField>

<ResponseField name="total_monthly_spend" type="integer">
Current monthly spend in the smallest currency unit.
</ResponseField>

<ResponseField name="active_plans" type="integer">
Number of active plan subscriptions.
</ResponseField>

<ResponseField name="active_deployments" type="integer">
Number of active deployment subscriptions.
</ResponseField>

<ResponseField name="has_payment_method" type="boolean">
Whether a payment method is on file.
</ResponseField>

<ResponseField name="available_credit" type="integer">
Available credit in the smallest currency unit.
</ResponseField>

<ResponseField name="outstanding_balance" type="integer">
Outstanding balance in the smallest currency unit.
</ResponseField>

<ResponseField name="subscriptions" type="array">
Active subscriptions.
<Expandable>
  <ResponseField name="id" type="string">
  Stripe subscription id.
  </ResponseField>
  <ResponseField name="status" type="string">
  Subscription status, such as `active`.
  </ResponseField>
  <ResponseField name="plan" type="string">
  Plan code, such as `DEPLOY_DEV` or `PRODUCTION_MONTHLY`.
  </ResponseField>
  <ResponseField name="plan_name" type="string">
  Display name, such as `Sandbox Deployment` or `Production (Monthly)`.
  </ResponseField>
  <ResponseField name="type" type="string">
  `plan` or `deployment`.
  </ResponseField>
  <ResponseField name="quantity" type="integer">
  Number of units on the subscription.
  </ResponseField>
  <ResponseField name="unit_amount" type="integer">
  Price per unit in the smallest currency unit.
  </ResponseField>
  <ResponseField name="currency" type="string">
  ISO currency code, such as `usd`.
  </ResponseField>
  <ResponseField name="interval" type="string">
  Billing interval, such as `month`.
  </ResponseField>
  <ResponseField name="current_period_start" type="integer">
  Unix timestamp for the start of the current period.
  </ResponseField>
  <ResponseField name="current_period_end" type="integer">
  Unix timestamp for the end of the current period.
  </ResponseField>
  <ResponseField name="cancel_at_period_end" type="boolean">
  Whether the subscription cancels at the end of the period.
  </ResponseField>
</Expandable>
</ResponseField>
